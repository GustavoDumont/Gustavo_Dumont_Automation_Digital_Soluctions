📋 Estrutura da planilha gerada:
• Linha 1: Instrução sobre macro para ocultar colunas
• Linha 2: Cabeçalho — Nome, CPF, Cargo, Situação, CC + uma coluna por treinamento
• Linha 3+: Dados com data de realização e vencimento por treinamento
• Cores: Verde = Em dia, Vermelho = Vencido, Amarelo = A vencer, Cinza = Sem data
• Macros VBA: Copie o código abaixo no Editor VBA para ocultar/exibir colunas vazias
' ── Cole este código no Editor VBA do Excel (Alt+F11) ──
Sub OcultarColunaVazia()
  Dim ws As Worksheet, c As Long, tem As Boolean
  Set ws = ActiveSheet
  Application.ScreenUpdating = False
  For c = 6 To ws.Cells(2, ws.Columns.Count).End(xlToLeft).Column
    tem = False
    Dim i As Long
    For i = 3 To ws.Cells(ws.Rows.Count,1).End(xlUp).Row
      If ws.Cells(i, c).Value <> "" Then tem = True: Exit For
    Next i
    ws.Columns(c).Hidden = Not tem
  Next c
  Application.ScreenUpdating = True
End Sub

Sub ReexibirTodas()
  ActiveSheet.Cells.EntireColumn.Hidden = False
End Sub
